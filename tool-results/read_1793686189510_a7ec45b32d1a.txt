     1→import { z } from 'zod';
     2→import { ApiError } from '../../lib/http';
     3→
     4→/**
     5→ * Master-data / capacity resource registry.
     6→ *
     7→ * A single declarative description per entity drives the generic CRUD,
     8→ * search, CSV export and CSV import endpoints — so every master-data screen
     9→ * gets the same behaviour (جستجو، خروجی و ورودی اکسل) without duplicated code.
    10→ */
    11→
    12→export const decimalField = z
    13→  .union([z.number(), z.string()])
    14→  .transform((v) => String(v))
    15→  .nullable()
    16→  .optional();
    17→
    18→export const optionalTrimmed = (max: number) =>
    19→  z
    20→    .string()
    21→    .trim()
    22→    .max(max)
    23→    .transform((v) => (v.length === 0 ? null : v))
    24→    .nullable()
    25→    .optional();
    26→
    27→export const requiredCode = (max: number) => z.string().trim().min(1).max(max);
    28→
    29→// ---------------------------------------------------------------------------
    30→// column helpers — one definition drives both table UI and CSV export
    31→// ---------------------------------------------------------------------------
    32→
    33→export interface CsvColumn {
    34→  /** Persian Excel header */
    35→  header: string;
    36→  /** row key for scalar columns (used when `get` is absent) */
    37→  key?: string;
    38→  /** custom value for export; return string | number | null */
    39→  get?: (row: any) => string | number | null;
    40→}
    41→
    42→export interface ImportColumn {
    43→  /** CSV header name (Persian) — matching is trimmed + whitespace-insensitive */
    44→  header: string;
    45→  /** target key in the prisma data object */
    46→  key: string;
    47→  kind?: 'string' | 'int' | 'decimal' | 'boolean' | 'enum';
    48→  /** allowed enum values (kind = enum) */
    49→  values?: readonly string[];
    50→  /** required in every imported row */
    51→  required?: boolean;
    52→  /** resolve external reference (returns id or throws Persian error) */
    53→  ref?: 'unit' | 'material' | 'part' | 'machine' | 'operation' | 'die' | 'warehouse' | 'customer';
    54→}
    55→
    56→export interface ResourceDef {
    57→  key: string;
    58→  /** prisma delegate name */
    59→  model: string;
    60→  /** permission module used for view/manage guards */
    61→  permModule: string;
    62→  labelFa: string;
    63→  uniqueField: string;
    64→  /** free-text search across these scalar fields (contains, insensitive) */
    65→  searchFields: string[];
    66→  /** free-text search inside relations, e.g. { part: ['partNumber','nameFa'] } */
    67→  searchRelations?: Record<string, string[]>;
    68→  /** exact-match query filters allowed (field → optional value map) */
    69→  filters: string[];
    70→  /** enum filters: { [queryKey]: allowedValues } — validated before query */
    71→  enumFilters?: Record<string, readonly string[]>;
    72→  orderBy: Record<string, 'asc' | 'desc'>;
    73→  /** prisma include for list/get (kept small) */
    74→  include?: Record<string, unknown>;
    75→  /** extra relation filter applied to list (e.g. location belongs to warehouse) */
    76→  create: z.ZodTypeAny;
    77→  update: z.ZodTypeAny;
    78→  /** CSV export columns */
    79→  columns: CsvColumn[];
    80→  /** CSV import mapping — omit for resources without import (e.g. joins handled elsewhere) */
    81→  import?: { columns: ImportColumn[]; example: Record<string, string> };
    82→  /** delete strategy */
    83→  remove: 'soft' | 'hard';
    84→  /** for soft: field + inactive value */
    85→  soft?: { field: 'status' | 'isActive'; value: unknown; activeWhere?: Record<string, unknown> };
    86→}

   794→}
