    41→/** Part-level WIP aggregates over ACTIVE MOs (same formulas as the WIP dashboard). */
    42→async function wipByPart(partId?: string, operationId?: string): Promise<Map<string, { waiting: D; processing: D; ready: D; total: D; scrap: D; rejected: D; maxAgingHours: number | null }>> {
    43→  const ops = await db.productionOperation.findMany({
    44→    where: {
    45→      manufacturingOrder: { status: { in: ACTIVE_MO_STATUSES }, ...(partId ? { partId } : {}) },
    46→      ...(operationId ? { operationId } : {}),
    47→    },
    48→    select: {
    49→      receivedQuantity: true, inputQuantity: true, producedQuantity: true,
    50→      scrappedQuantity: true, rejectedQuantity: true, reworkQuantity: true, transferredOutQuantity: true,
    51→      wipSinceAt: true,
    52→      manufacturingOrder: { select: { partId: true } },
    53→    },
    54→  });
    55→  const map = new Map<string, { waiting: D; processing: D; ready: D; total: D; scrap: D; rejected: D; maxAgingHours: number | null }>();
    56→  const now = Date.now();
    57→  for (const op of ops) {
    58→    const pid = op.manufacturingOrder.partId;
    59→    const acc = map.get(pid) ?? { waiting: ZERO, processing: ZERO, ready: ZERO, total: ZERO, scrap: ZERO, rejected: ZERO, maxAgingHours: null };
    60→    const received = Q(op.receivedQuantity);
    61→    const input = Q(op.inputQuantity);
    62→    const produced = Q(op.producedQuantity);
    63→    const scrap = Q(op.scrappedQuantity);
    64→    const rejected = Q(op.rejectedQuantity);
    65→    const rework = Q(op.reworkQuantity);
    66→    const transferredOut = Q(op.transferredOutQuantity);
    67→    acc.waiting = acc.waiting.plus(received).plus(rework).minus(input);
    68→    acc.processing = acc.processing.plus(input).minus(produced).minus(scrap).minus(rejected).minus(rework);
    69→    acc.ready = acc.ready.plus(produced).minus(transferredOut);
    70→    acc.total = acc.total.plus(received).minus(transferredOut).minus(scrap).minus(rejected);
    71→    acc.scrap = acc.scrap.plus(scrap);
    72→    acc.rejected = acc.rejected.plus(rejected);
    73→    if (op.wipSinceAt) {
    74→      const h = Math.max(0, (now - op.wipSinceAt.getTime()) / 3_600_000);
    75→      acc.maxAgingHours = acc.maxAgingHours === null ? h : Math.max(acc.maxAgingHours, h);
    76→    }
    77→    map.set(pid, acc);
    78→  }
    79→  return map;
    80→}

    82→/** Usable FG stock per part (OWNED + APPROVED in FG warehouses) minus ACTIVE reservations. */
    83→async function fgAvailableByPart(partId?: string): Promise<Map<string, { usable: D; consignment: D; quarantined: D }>> {
    84→  const fgIds = await fgWarehouseIds();
    85→  if (fgIds.length === 0) return new Map();
    86→  const [balances, reservations] = await Promise.all([
    87→    db.stockBalance.groupBy({
    88→      by: ['partId', 'ownership', 'qualityStatus'],
    89→      _sum: { quantity: true },
    90→      where: { warehouseId: { in: fgIds }, partId: { not: null }, ...(partId ? { partId } : {}) },
    91→    }),
    92→    db.inventoryReservation.groupBy({
    93→      by: ['partId'],
    94→      _sum: { quantity: true },
    95→      where: { status: 'ACTIVE', partId: { not: null }, ...(partId ? { partId } : {}) },
    96→    }),
    97→  ]);
    98→  const usable = new Map<string, D>();
    99→  const consignment = new Map<string, D>();
   100→  const quarantined = new Map<string, D>();
   101→  for (const b of balances) {
   102→    if (!b.partId) continue;
   103→    if (b.ownership === 'OWNED' && b.qualityStatus === 'APPROVED') {
   104→      usable.set(b.partId, (usable.get(b.partId) ?? ZERO).plus(Q(b._sum.quantity)));
   105→    } else if (b.ownership === 'SUPPLIER_CONSIGNMENT' && b.qualityStatus === 'APPROVED') {
   106→      consignment.set(b.partId, (consignment.get(b.partId) ?? ZERO).plus(Q(b._sum.quantity)));
   107→    } else {
   108→      quarantined.set(b.partId, (quarantined.get(b.partId) ?? ZERO).plus(Q(b._sum.quantity)));
   109→    }
   110→  }
   111→  const reserved = new Map<string, D>();
   112→  for (const r of reservations) if (r.partId) reserved.set(r.partId, Q(r._sum.quantity));
   113→  const out = new Map<string, { usable: D; consignment: D; quarantined: D }>();
   114→  const keys = new Set([...usable.keys(), ...consignment.keys(), ...quarantined.keys(), ...reserved.keys()]);
   115→  for (const k of keys) {
   116→    out.set(k, {
   117→      usable: (usable.get(k) ?? ZERO).minus(reserved.get(k) ?? ZERO),
   118→      consignment: consignment.get(k) ?? ZERO,
   119→      quarantined: quarantined.get(k) ?? ZERO,
   120→    });
   121→  }
   122→  return out;
   123→}

   129→export async function planVsActual(f: { from?: string; to?: string; partId?: string; customerId?: string }): Promise<ReportTable> {
   130→  const plans = await db.productionPlan.findMany({
   131→    where: {
   132→      status: { in: ['CONFIRMED', 'RELEASED', 'CLOSED'] },
   133→      ...(f.from ? { periodStart: { gte: new Date(`${f.from}T00:00:00.000Z`) } } : {}),
   134→      ...(f.to ? { periodEnd: { lte: endOfDay(new Date(`${f.to}T00:00:00.000Z`)) } } : {}),
   135→    },
   136→    select: { id: true, planNumber: true, version: true, status: true, periodStart: true, periodEnd: true },
   137→    orderBy: { periodStart: 'desc' },
   138→    take: 200,
   139→  });
   140→  const planIds = plans.map((p) => p.id);
   141→  const lines = planIds.length
   142→    ? await db.productionPlanLine.findMany({
   143→        where: { planId: { in: planIds }, ...(f.partId ? { partId: f.partId } : {}) },
   144→        include: {
   145→          part: { select: { partNumber: true, nameFa: true } },
   146→          plan: { select: { planNumber: true, version: true, status: true } },
   147→          customerOrderLine: { select: { order: { select: { customerId: true, orderNumber: true } } } },
   148→          manufacturingOrders: { select: { id: true, moNumber: true, status: true, quantity: true, operations: { orderBy: { sequence: 'desc' }, take: 1, select: { producedQuantity: true } } } },
   149→        },
   150→        orderBy: { lineNo: 'asc' },
   151→        take: 1000,
   152→      })
   153→    : [];
   154→
   155→  const rows: Record<string, unknown>[] = [];
   156→  for (const l of lines) {
   157→    if (f.customerId && l.customerOrderLine?.order.customerId !== f.customerId) continue;
   158→    const planned = Q(l.plannedQuantity);
   159→    const moQty = l.manufacturingOrders.reduce((acc, mo) => acc.plus(Q(mo.quantity)), ZERO);
   160→    // ACTUAL: final-operation produced quantity aggregated by confirmed production reports
   161→    const actual = l.manufacturingOrders.reduce((acc, mo) => acc.plus(mo.operations[0] ? Q(mo.operations[0].producedQuantity) : ZERO), ZERO);
   162→    rows.push({
   163→      planNumber: l.plan.planNumber,
   164→      planVersion: l.plan.version,
   165→      planStatus: l.plan.status,
   166→      lineNo: l.lineNo,
   167→      part: `${l.part.partNumber} — ${l.part.nameFa}`,
   168→      orderNumber: l.customerOrderLine?.order.orderNumber ?? '—',
   169→      plannedQuantity: qty(planned),
   170→      coveredByStock: qty(l.coveredByStock),
   171→      coveredByReservations: qty(l.coveredByReservations),
   172→      coveredByWip: qty(l.coveredByWip),
   173→      coveredByPlannedMos: qty(l.coveredByPlannedMos),
   174→      moCreatedQuantity: qty(moQty),
   175→      actualQuantity: qty(actual),
   176→      variance: qty(actual.minus(planned)),
   177→      variancePct: planned.comparedTo(0) !== 0 ? Math.round(actual.minus(planned).div(planned).mul(100).toNumber() * 10) / 10 : null,
   178→      dueDate: l.dueDate ? isoDate(l.dueDate) : null,
   179→      _drill: drill('plan_line', l.id, `ردیف ${l.lineNo} برنامه ${l.plan.planNumber}`),
   180→    });
   181→  }
   182→
   183→  const columns: ReportColumn[] = [
   184→    { key: 'planNumber', labelFa: 'شماره برنامه', kind: 'IDENTIFIER' },
   185→    { key: 'planVersion', labelFa: 'نسخه', kind: 'IDENTIFIER' },
   186→    { key: 'planStatus', labelFa: 'وضعیت برنامه', kind: 'INFO' },
   187→    { key: 'lineNo', labelFa: 'ردیف', kind: 'IDENTIFIER' },
   188→    { key: 'part', labelFa: 'قطعه', kind: 'IDENTIFIER' },
   189→    { key: 'orderNumber', labelFa: 'سفارش مشتری', kind: 'IDENTIFIER' },
   190→    { key: 'plannedQuantity', labelFa: 'برنامه‌ریزی‌شده', kind: 'PLANNED', unit: 'عدد' },
   191→    { key: 'coveredByStock', labelFa: 'پوشش موجودی', kind: 'PLANNED', unit: 'عدد' },
   192→    { key: 'coveredByReservations', labelFa: 'پوشش رزرو', kind: 'PLANNED', unit: 'عدد' },
   193→    { key: 'coveredByWip', labelFa: 'پوشش WIP', kind: 'PLANNED', unit: 'عدد' },
   194→    { key: 'coveredByPlannedMos', labelFa: 'پوشش دستور کار باز', kind: 'PLANNED', unit: 'عدد' },
   195→    { key: 'moCreatedQuantity', labelFa: 'دستور کار ایجادشده', kind: 'PLANNED', unit: 'عدد' },
   196→    { key: 'actualQuantity', labelFa: 'تولید واقعی (آخرین عملیات)', kind: 'ACTUAL', unit: 'عدد' },
   197→    { key: 'variance', labelFa: 'انحراف', kind: 'INFO', unit: 'عدد' },
   198→    { key: 'variancePct', labelFa: 'انحراف ٪', kind: 'INFO' },
   199→    { key: 'dueDate', labelFa: 'سررسید', kind: 'INFO' },
   200→    { key: '_drill', labelFa: 'جزئیات منبع', kind: 'INFO' },
   201→  ];
   202→
   203→  return table('plan-vs-actual', 'برنامه تولید در مقابل عملکرد واقعی', columns, rows, meta(f, [
   204→    'تولید واقعی = مجموع تولید آخرین عملیات دستور کارهای مرتبط با ردیف برنامه (فقط گزارش‌های تأییدشده که در تراکنش دیتابیس تجمیع شده‌اند).',
   205→    'مقادیر «پوشش» خروجی موتور برنامه‌ریزی در زمان ساخت برنامه‌اند و با وضعیت فعلی موجودی/WIP به‌روزرسانی نمی‌شوند.',
   206→  ]));
   207→}

   213→export async function orderFulfillment(f: { from?: string; to?: string; partId?: string; customerId?: string }): Promise<ReportTable> {
   214→  const lines = await db.customerOrderLine.findMany({
   215→    where: {
   216→      partId: f.partId,
   217→      order: {
   218→        customerId: f.customerId,
   219→        status: { notIn: ['CANCELLED'] },
   220→        ...(f.from ? { orderDate: { gte: new Date(`${f.from}T00:00:00.000Z`) } } : {}),
   221→        ...(f.to ? { orderDate: { lte: endOfDay(new Date(`${f.to}T00:00:00.000Z`)) } } : {}),
   222→      },
   223→    },
   224→    include: {
   225→      order: { select: { orderNumber: true, orderDate: true, status: true, customer: { select: { nameFa: true } } } },
   226→      part: { select: { partNumber: true, nameFa: true } },
   227→    },
   228→    orderBy: [{ order: { orderDate: 'desc' } }, { lineNo: 'asc' }],
   229→    take: 1000,
   230→  });
   231→
   232→  const today = new Date(`${isoDate(new Date())}T00:00:00.000Z`);
   233→  const rows = lines.map((l) => {
   234→    const ordered = Q(l.quantity);
   235→    const delivered = Q(l.deliveredQuantity);
   236→    const open = ordered.minus(delivered);
   237→    const overdue = open.comparedTo(0) > 0 && l.dueDate !== null && l.dueDate.getTime() < today.getTime();
   238→    return {
   239→      orderNumber: l.order.orderNumber,
   240→      customer: l.order.customer.nameFa,
   241→      orderStatus: l.order.status,
   242→      lineNo: l.lineNo,
   243→      part: `${l.part.partNumber} — ${l.part.nameFa}`,
   244→      orderDate: isoDate(l.order.orderDate),
   245→      dueDate: l.dueDate ? isoDate(l.dueDate) : null,
   246→      orderedQuantity: qty(ordered),
   247→      deliveredQuantity: qty(delivered),
   248→      openQuantity: qty(open),
   249→      fulfillmentPct: ordered.comparedTo(0) !== 0 ? Math.round(delivered.div(ordered).mul(100).toNumber() * 10) / 10 : null,
   250→      lineStatus: l.status,
   251→      overdue,
   252→      _drill: drill('order_line', l.id, `ردیف ${l.lineNo} سفارش ${l.order.orderNumber}`),
   253→    };
   254→  });
   255→
   256→  const columns: ReportColumn[] = [
   257→    { key: 'orderNumber', labelFa: 'شماره سفارش', kind: 'IDENTIFIER' },
   258→    { key: 'customer', labelFa: 'مشتری', kind: 'IDENTIFIER' },
   259→    { key: 'orderStatus', labelFa: 'وضعیت سفارش', kind: 'INFO' },
   260→    { key: 'lineNo', labelFa: 'ردیف', kind: 'IDENTIFIER' },
   261→    { key: 'part', labelFa: 'قطعه', kind: 'IDENTIFIER' },
   262→    { key: 'orderDate', labelFa: 'تاریخ سفارش', kind: 'INFO' },
   263→    { key: 'dueDate', labelFa: 'سررسید', kind: 'INFO' },
   264→    { key: 'orderedQuantity', labelFa: 'مقدار سفارش', kind: 'ACTUAL', unit: 'عدد' },
   265→    { key: 'deliveredQuantity', labelFa: 'تحویل‌شده', kind: 'ACTUAL', unit: 'عدد' },
   266→    { key: 'openQuantity', labelFa: 'باقی‌مانده', kind: 'INFO', unit: 'عدد' },
   267→    { key: 'fulfillmentPct', labelFa: 'اقلا ٪', kind: 'INFO' },
   268→    { key: 'lineStatus', labelFa: 'وضعیت ردیف', kind: 'INFO' },
   269→    { key: 'overdue', labelFa: 'سررسید گذشته', kind: 'INFO' },
   270→    { key: '_drill', labelFa: 'جزئیات منبع', kind: 'INFO' },
   271→  ];
   272→
   273→  return table('order-fulfillment', 'اقلای سفارش‌های مشتری', columns, rows, meta(f, [
   274→    'تحویل‌شده از فیلد واقعی ردیف سفارش خوانده می‌شود؛ سامانه تاریخ تحویل ثبت نمی‌کند، بنابراین «تحویل در سررسید» محاسبه نمی‌شود (عدم فرض ساختگی).',
   275→    '«سررسید گذشته» یعنی باقی‌مانده‌ای وجود دارد و تاریخ سررسید قبل از امروز است.',
   276→  ]));
   277→}

   283→export async function backlog(f: { partId?: string; customerId?: string }): Promise<ReportTable> {
   284→  const orderLines = await db.customerOrderLine.findMany({
   285→    where: {
   286→      status: { in: OPEN_LINE_STATUSES },
   287→      partId: f.partId,
   288→      order: { status: { in: OPEN_ORDER_STATUSES }, customerId: f.customerId },
   289→    },
   290→    select: { partId: true, quantity: true, deliveredQuantity: true, dueDate: true },
   291→  });
   292→  const openMos = await db.manufacturingOrder.groupBy({
   293→    by: ['partId'],
   294→    _sum: { quantity: true },
   295→    where: { status: { in: OPEN_MO_STATUSES }, partId: f.partId, ...(f.customerId ? { planLine: { customerOrderLine: { order: { customerId: f.customerId } } } } : {}) },
   296→  });
   297→
   298→  const demandByPart = new Map<string, D>();
   299→  const earliestDue = new Map<string, Date | null>();
   300→  for (const l of orderLines) {
   301→    const open = Q(l.quantity).minus(Q(l.deliveredQuantity));
   302→    if (open.comparedTo(0) <= 0) continue;
   303→    demandByPart.set(l.partId, (demandByPart.get(l.partId) ?? ZERO).plus(open));
   304→    const cur = earliestDue.get(l.partId);
   305→    if (l.dueDate && (cur === undefined || cur === null || l.dueDate.getTime() < cur.getTime())) earliestDue.set(l.partId, l.dueDate);
   306→    else if (!earliestDue.has(l.partId)) earliestDue.set(l.partId, null);
   307→  }
   308→
   309→  const [wip, fg] = await Promise.all([wipByPart(f.partId), fgAvailableByPart(f.partId)]);
   310→  const moByPart = new Map<string, D>();
   311→  for (const m of openMos) moByPart.set(m.partId, Q(m._sum.quantity));
   312→
   313→  const parts = await db.part.findMany({
   314→    where: { id: { in: [...demandByPart.keys(), ...moByPart.keys()].filter((x) => !f.customerId || f.customerId) } },
   315→    select: { id: true, partNumber: true, nameFa: true },
   316→  });
   317→  const partById = new Map(parts.map((p) => [p.id, p]));
   318→
   319→  const rows: Record<string, unknown>[] = [];
   320→  const keys = new Set<string>([...demandByPart.keys(), ...moByPart.keys()]);
   321→  for (const pid of keys) {
   322→    const part = partById.get(pid);
   323→    if (!part) continue;
   324→    const openOrders = demandByPart.get(pid) ?? ZERO;
   325→    const w = wip.get(pid);
   326→    const g = fg.get(pid);
   327→    const openMo = moByPart.get(pid) ?? ZERO;
   328→    const wipTotal = w?.total ?? ZERO;
   329→    const fgUsable = g?.usable ?? ZERO;
   330→    const coverage = fgUsable.plus(wipTotal).plus(openMo);
   331→    rows.push({
   332→      part: `${part.partNumber} — ${part.nameFa}`,
   333→      openOrderQuantity: qty(openOrders),
   334→      earliestDueDate: earliestDue.has(pid) && earliestDue.get(pid) ? isoDate(earliestDue.get(pid)!) : null,
   335→      openMoQuantity: qty(openMo),
   336→      wipQuantity: qty(wipTotal),
   337→      finishedGoodsUsable: qty(fgUsable),
   338→      finishedGoodsConsignment: qty(g?.consignment ?? ZERO),
   339→      coverageQuantity: qty(coverage),
   340→      uncoveredBacklog: qty(openOrders.minus(coverage)),
   341→      coveredFully: openOrders.minus(coverage).comparedTo(0) <= 0,
   342→    });
   343→  }
   344→  rows.sort((a, b) => String(a.part).localeCompare(String(b.part), 'fa'));
   345→
   346→  const columns: ReportColumn[] = [
   347→    { key: 'part', labelFa: 'قطعه', kind: 'IDENTIFIER' },
   348→    { key: 'openOrderQuantity', labelFa: 'تقاضای باز سفارش‌ها', kind: 'ACTUAL', unit: 'عدد' },
   349→    { key: 'earliestDueDate', labelFa: 'نزدیک‌ترین سررسید', kind: 'INFO' },
   350→    { key: 'openMoQuantity', labelFa: 'دستور کار باز', kind: 'ACTUAL', unit: 'عدد' },
   351→    { key: 'wipQuantity', labelFa: 'WIP (کل)', kind: 'ACTUAL', unit: 'عدد' },
   352→    { key: 'finishedGoodsUsable', labelFa: 'موجودی قابل استفاده محصول', kind: 'ACTUAL', unit: 'عدد' },
   353→    { key: 'finishedGoodsConsignment', labelFa: 'موجودی امانی (غیرقابل استفاده)', kind: 'ACTUAL', unit: 'عدد' },
   354→    { key: 'coverageQuantity', labelFa: 'پوشش کل', kind: 'INFO', unit: 'عدد' },
   355→    { key: 'uncoveredBacklog', labelFa: 'کار پوشش‌نیافته', kind: 'INFO', unit: 'عدد' },
   356→    { key: 'coveredFully', labelFa: 'پوشش کامل؟', kind: 'INFO' },
   357→  ];
   358→
   359→  return table('backlog', 'کار در انتظار تولید (Backlog)', columns, rows, meta(f, [
   360→    'موجودی قابل استفاده = انبارهای محصول تکمیلی با مالکیت مالی و تأیید کیفیت منهای رزروهای فعال (منبع: موجودی رسمی).',

   369→export async function wipByPartOperation(f: { partId?: string; operationId?: string }): Promise<ReportTable> {
   370→  const ops = await db.productionOperation.findMany({
   371→    where: {
   372→      manufacturingOrder: { status: { in: ACTIVE_MO_STATUSES }, partId: f.partId },
   373→      operationId: f.operationId,
   374→    },
   375→    include: {
   376→      operation: { select: { code: true, nameFa: true } },
   377→      manufacturingOrder: { select: { moNumber: true, partId: true, part: { select: { partNumber: true, nameFa: true } } } },
   378→    },
   379→    orderBy: [{ manufacturingOrder: { moNumber: 'asc' } }, { sequence: 'asc' }],
   380→    take: 2000,
   381→  });
   382→
   383→  const group = new Map<string, {
   384→    part: string; partId: string; operationCode: string; operationName: string;
   385→    waiting: D; processing: D; ready: D; total: D; ops: number; maxAging: number | null;
   386→  }>();
   387→  for (const op of ops) {
   388→    const key = `${op.manufacturingOrder.part.partNumber}|${op.operation.code}`;
   389→    const g = group.get(key) ?? {
   390→      part: `${op.manufacturingOrder.part.partNumber} — ${op.manufacturingOrder.part.nameFa}`,
   391→      partId: op.manufacturingOrder.partId,
   392→      operationCode: op.operation.code,
   393→      operationName: op.operation.nameFa,
   394→      waiting: ZERO, processing: ZERO, ready: ZERO, total: ZERO, ops: 0, maxAging: null as number | null,
   395→    };
   396→    const received = Q(op.receivedQuantity);
   397→    const input = Q(op.inputQuantity);
   398→    const produced = Q(op.producedQuantity);
   399→    const scrap = Q(op.scrappedQuantity);
   400→    const rejected = Q(op.rejectedQuantity);
   401→    const rework = Q(op.reworkQuantity);
   402→    const out = Q(op.transferredOutQuantity);
   403→    g.waiting = g.waiting.plus(received).plus(rework).minus(input);
   404→    g.processing = g.processing.plus(input).minus(produced).minus(scrap).minus(rejected).minus(rework);
   405→    g.ready = g.ready.plus(produced).minus(out);
   406→    g.total = g.total.plus(received).minus(out).minus(scrap).minus(rejected);
   407→    g.ops += 1;
   408→    if (op.wipSinceAt) {
   409→      const h = Math.max(0, (Date.now() - op.wipSinceAt.getTime()) / 3_600_000);
   410→      g.maxAging = g.maxAging === null ? h : Math.max(g.maxAging, h);
   411→    }
   412→    group.set(key, g);
   413→  }
   414→
   415→  const rows = [...group.values()]
   416→    .sort((a, b) => a.part.localeCompare(b.part, 'fa') || a.operationCode.localeCompare(b.operationCode))
   417→    .map((g) => ({
   418→      part: g.part,
   419→      operation: `${g.operationCode} — ${g.operationName}`,
   420→      operationCount: g.ops,
   421→      waitingToProcess: qty(g.waiting),
   422→      processing: qty(g.processing),
   423→      readyForTransfer: qty(g.ready),
   424→      wipTotal: qty(g.total),
   425→      maxAgingHours: g.maxAging === null ? null : Math.round(g.maxAging * 10) / 10,
   426→      conservationCheck: qty(g.waiting.plus(g.processing).plus(g.ready)),
   427→      _drill: drill('wip_part_operation', `${g.partId}|${g.operationCode}`, `${g.part} / ${g.operationCode}`),
   428→    }));
   429→
   430→  const columns: ReportColumn[] = [
   431→    { key: 'part', labelFa: 'قطعه', kind: 'IDENTIFIER' },
   432→    { key: 'operation', labelFa: 'عملیات', kind: 'IDENTIFIER' },
   433→    { key: 'operationCount', labelFa: 'تعداد عملیات فعال', kind: 'INFO' },
   434→    { key: 'waitingToProcess', labelFa: 'در انتظار پردازش', kind: 'ACTUAL', unit: 'عدد' },
   435→    { key: 'processing', labelFa: 'در جریان پردازش', kind: 'ACTUAL', unit: 'عدد' },
   436→    { key: 'readyForTransfer', labelFa: 'آماده انتقال', kind: 'ACTUAL', unit: 'عدد' },
   437→    { key: 'wipTotal', labelFa: 'جمع WIP', kind: 'ACTUAL', unit: 'عدد' },
   438→    { key: 'maxAgingHours', labelFa: 'قداره حداکثری (ساعت)', kind: 'ACTUAL' },
   439→    { key: 'conservationCheck', labelFa: 'کنترل حفظ کمیت (انتظار = جمع WIP)', kind: 'INFO', unit: 'عدد' },
   440→    { key: '_drill', labelFa: 'جزئیات منبع', kind: 'INFO' },
   441→  ];
   442→
   443→  return table('wip-by-part-operation', 'قطعات در جریان ساخت به تفکیک قطعه و عملیات', columns, rows, meta(f, [
   444→    'فرمول‌ها با داشبورد WIP یکسان است: در انتظار = رسیده + بازکار − تغذیه؛ در جریان = تغذیه − تولید − ضایعات − مردود − بازکار؛ آماده انتقال = تولید − منتقل‌شده.',
   445→  ]));
   446→}

   449→// 5) WIP aging
   450→// ---------------------------------------------------------------------------
   451→
   452→export async function wipAging(f: { partId?: string; operationId?: string }): Promise<ReportTable> {
   453→  const ops = await db.productionOperation.findMany({
   454→    where: {
   455→      manufacturingOrder: { status: { in: ACTIVE_MO_STATUSES }, partId: f.partId },
   456→      operationId: f.operationId,
   457→    },
   458→    include: {
   459→      operation: { select: { code: true, nameFa: true } },
   460→      manufacturingOrder: { select: { moNumber: true, part: { select: { partNumber: true, nameFa: true } } } },
   461→      machine: { select: { code: true } },
   462→    },
   463→    orderBy: { wipSinceAt: 'asc' },
   464→    take: 2000,
   465→  });
   466→
   467→  const now = Date.now();
   468→  const buckets = { under24h: 0, from24to72h: 0, from72to168h: 0, over168h: 0, unknown: 0 };
   469→  const rows: Record<string, unknown>[] = [];
   470→  for (const op of ops) {
   471→    const received = Q(op.receivedQuantity);
   472→    const input = Q(op.inputQuantity);
   473→    const produced = Q(op.producedQuantity);
   474→    const scrap = Q(op.scrappedQuantity);
   475→    const rejected = Q(op.rejectedQuantity);
   476→    const rework = Q(op.reworkQuantity);
   477→    const out = Q(op.transferredOutQuantity);
   478→    const wipTotal = received.minus(out).minus(scrap).minus(rejected);
   479→    if (wipTotal.comparedTo(0) <= 0) continue;
   480→
   481→    let agingHours: number | null = null;
   482→    let bucket = 'unknown';
   483→    if (op.wipSinceAt) {
   484→      agingHours = Math.max(0, (now - op.wipSinceAt.getTime()) / 3_600_000);
   485→      if (agingHours < 24) { bucket = 'under24h'; buckets.under24h += 1; }
   486→      else if (agingHours < 72) { bucket = 'from24to72h'; buckets.from24to72h += 1; }
   487→      else if (agingHours < 168) { bucket = 'from72to168h'; buckets.from72to168h += 1; }
   488→      else { bucket = 'over168h'; buckets.over168h += 1; }
   489→    } else {
   490→      buckets.unknown += 1;
   491→    }
   492→    rows.push({
   493→      moNumber: op.manufacturingOrder.moNumber,
   494→      part: `${op.manufacturingOrder.part.partNumber} — ${op.manufacturingOrder.part.nameFa}`,
   495→      sequence: op.sequence,
   496→      operation: `${op.operation.code} — ${op.operation.nameFa}`,
   497→      machine: op.machine?.code ?? null,
   498→      wipTotal: qty(wipTotal),
   499→      readyForTransfer: qty(produced.minus(out)),
   500→      waitingToProcess: qty(received.plus(rework).minus(input)),
   501→      agingHours: agingHours === null ? null : Math.round(agingHours * 10) / 10,
   502→      agingBucket: bucket,
   503→      agingKnown: agingHours !== null,
   504→      _drill: drill('wip_operation', op.id, `${op.manufacturingOrder.moNumber} / ${op.operation.code}`),
   505→    });
   506→  }
   507→
   508→  const columns: ReportColumn[] = [
   509→    { key: 'moNumber', labelFa: 'دستور کار', kind: 'IDENTIFIER' },
   510→    { key: 'part', labelFa: 'قطعه', kind: 'IDENTIFIER' },
   511→    { key: 'sequence', labelFa: 'ترتیب', kind: 'IDENTIFIER' },
   512→    { key: 'operation', labelFa: 'عملیات', kind: 'IDENTIFIER' },
   513→    { key: 'machine', labelFa: 'ماشین', kind: 'IDENTIFIER' },
   514→    { key: 'wipTotal', labelFa: 'جمع WIP', kind: 'ACTUAL', unit: 'عدد' },
   515→    { key: 'readyForTransfer', labelFa: 'آماده انتقال', kind: 'ACTUAL', unit: 'عدد' },
   516→    { key: 'waitingToProcess', labelFa: 'در انتظار پردازش', kind: 'ACTUAL', unit: 'عدد' },
   517→    { key: 'agingHours', labelFa: 'قداره (ساعت)', kind: 'ACTUAL' },
   518→    { key: 'agingBucket', labelFa: 'بازه سنی', kind: 'INFO' },
   519→    { key: 'agingKnown', labelFa: 'قداره مشخص است؟', kind: 'INFO' },
   520→    { key: '_drill', labelFa: 'جزئیات منبع', kind: 'INFO' },
   521→  ];
   522→
   523→  const notes = [
   524→    'قداره از زمانی که WIP عملیات از صفر بزرگ شد (wip_since_at) محاسبه می‌شود.',
   525→    'ردیف‌هایی که مبنای قداره ندارند صریحاً علامت خورده‌اند (قداره نامشخص) — بدون فرض.',
   526→    `خلاصه سنی: زیر ۲۴ ساعت: ${buckets.under24h} | ۲۴ تا ۷۲ ساعت: ${buckets.from24to72h} | ۷۲ تا ۱۶۸ ساعت: ${buckets.from72to168h} | بیش از ۱۶۸ ساعت: ${buckets.over168h} | نامشخص: ${buckets.unknown}`,
   527→  ];
   528→
   529→  return table('wip-aging', 'قداره قطعات در جریان ساخت', columns, rows, meta(f, notes));
   530→}

   536→export async function materialConsumption(f: { from?: string; to?: string; materialId?: string; partId?: string }): Promise<ReportTable> {
   537→  const { period, defaultApplied } = resolvePeriod(f.from, f.to);
   538→  const occurredAt = { ...(period.from ? { gte: period.from } : {}), ...(period.to ? { lte: period.to } : {}) };
   539→  const dateWhere = Object.keys(occurredAt).length ? { occurredAt } : {};
   540→
   541→  // ACTUAL consumption: posted issue transactions (immutable ledger)
   542→  const consumption = await db.inventoryTransaction.groupBy({
   543→    by: ['materialId'],
   544→    _sum: { quantity: true, weightKg: true },
   545→    where: { materialId: { not: null }, type: { in: CONSUMPTION_TX }, ...(f.materialId ? { materialId: f.materialId } : {}), ...dateWhere },
   546→  });
   547→
   548→  // ACTUAL production basis: FG receipts of parts in the same period
   549→  const fgIds = await fgWarehouseIds();
   550→  const produced = fgIds.length
   551→    ? await db.inventoryTransaction.groupBy({
   552→        by: ['partId'],
   553→        _sum: { quantity: true },
   554→        where: { partId: { not: null }, type: 'RECEIPT', warehouseId: { in: fgIds }, ...(f.partId ? { partId: f.partId } : {}), ...dateWhere },
   555→      })
   556→    : [];
   557→  const producedByPart = new Map<string, D>();
   558→  for (const p of produced) if (p.partId) producedByPart.set(p.partId, Q(p._sum.quantity));
   559→
   560→  const bom = await db.partMaterialRequirement.findMany({
   561→    where: { partId: { in: [...producedByPart.keys()] }, ...(f.materialId ? { materialId: f.materialId } : {}) },
   562→    select: { partId: true, materialId: true, quantityPerUnit: true, weightKgPerUnit: true, part: { select: { partNumber: true } } },
   563→  });
   564→
   565→  const [materials, allParts] = await Promise.all([
   566→    db.material.findMany({ where: f.materialId ? { id: f.materialId } : {}, select: { id: true, materialCode: true, nameFa: true, baseUnit: { select: { code: true } } } }),
   567→    db.part.findMany({ where: { id: { in: [...producedByPart.keys()] } }, select: { id: true, partNumber: true, nameFa: true } }),
   568→  ]);
   569→  const matById = new Map(materials.map((m) => [m.id, m]));
   570→  const partById = new Map(allParts.map((p) => [p.id, p]));
   571→
   572→  // parts produced in the period WITHOUT any BOM row → surfaced, never assumed
   573→  const partsWithBom = new Set(bom.map((b) => b.partId));
   574→  const missingBomParts = [...producedByPart.keys()].filter((pid) => !partsWithBom.has(pid));
   575→
   576→  const standardByMaterial = new Map<string, { qty: D; kg: D | null; basisParts: string[] }>();
   577→  for (const b of bom) {
   578→    const producedQty = producedByPart.get(b.partId);
   579→    if (!producedQty) continue;
   580→    const cur = standardByMaterial.get(b.materialId) ?? { qty: ZERO, kg: null, basisParts: [] as string[] };
   581→    cur.qty = cur.qty.plus(producedQty.mul(Q(b.quantityPerUnit)));
   582→    if (b.weightKgPerUnit !== null) cur.kg = (cur.kg ?? ZERO).plus(producedQty.mul(Q(b.weightKgPerUnit)));
   583→    cur.basisParts.push(partById.get(b.partId)?.partNumber ?? b.partId);
   584→    standardByMaterial.set(b.materialId, cur);
   585→  }
   586→
   587→  const consumedParts: string[] = [...producedByPart.keys()]
   588→    .map((pid) => partById.get(pid)?.partNumber ?? pid)
   589→    .sort();
   590→
   591→  const keys = new Set<string>([...consumption.map((c) => c.materialId as string), ...standardByMaterial.keys()]);
   592→  const rows: Record<string, unknown>[] = [];
   593→  for (const mid of keys) {
   594→    const m = matById.get(mid);
   595→    if (!m) continue;
   596→    const cons = consumption.find((c) => c.materialId === mid);
   597→    const consSum = cons?._sum;
   598→    const actualQty = Q(consSum?.quantity ?? 0);
   599→    const actualKg = consSum?.weightKg == null ? null : Q(consSum.weightKg);
   600→    const std = standardByMaterial.get(mid);
   601→    const stdQty = std?.qty ?? null;
   602→    rows.push({
   603→      material: `${m.materialCode} — ${m.nameFa}`,
   604→      unit: m.baseUnit.code,
   605→      actualConsumedQuantity: qty(actualQty),
   606→      actualConsumedKg: actualKg === null ? null : qty(actualKg),
   607→      standardQuantity: stdQty === null ? null : qty(stdQty),
   608→      standardKg: std?.kg == null ? null : qty(std.kg),
   609→      varianceQuantity: stdQty === null ? null : qty(actualQty.minus(stdQty)),
   610→      varianceKg: std?.kg == null || actualKg === null ? null : qty(actualKg.minus(std.kg)),
   611→      standardBasisParts: std ? std.basisParts.join('، ') : null,
   612→      standardKnown: stdQty !== null,
   613→      _drill: drill('material_consumption_group', mid, `مصرف ${m.materialCode}`),
   614→    });
   615→  }
   616→  rows.sort((a, b) => String(a.material).localeCompare(String(b.material), 'fa'));
   617→
   618→  const columns: ReportColumn[] = [
   619→    { key: 'material', labelFa: 'ماده اولیه', kind: 'IDENTIFIER' },
   620→    { key: 'unit', labelFa: 'واحد', kind: 'IDENTIFIER' },
   621→    { key: 'actualConsumedQuantity', labelFa: 'مصرف واقعی (مقدار)', kind: 'ACTUAL' },
   622→    { key: 'actualConsumedKg', labelFa: 'مصرف واقعی (kg)', kind: 'ACTUAL', unit: 'kg' },
   623→    { key: 'standardQuantity', labelFa: 'سرانه استاندارد (مقدار)', kind: 'STANDARD' },
   624→    { key: 'standardKg', labelFa: 'سرانه استاندارد (kg)', kind: 'STANDARD', unit: 'kg' },
   625→    { key: 'varianceQuantity', labelFa: 'انحراف مقدار', kind: 'INFO' },
   626→    { key: 'varianceKg', labelFa: 'انحراف kg', kind: 'INFO', unit: 'kg' },
   627→    { key: 'standardBasisParts', labelFa: 'مبنای استاندارد (قطعات تولیدی)', kind: 'INFO' },
   628→    { key: 'standardKnown', labelFa: 'سرانه قابل محاسبه؟', kind: 'INFO' },
   629→    { key: '_drill', labelFa: 'جزئیات منبع', kind: 'INFO' },
   630→  ];
   631→
   632→  const notes = [
   633→    'مصرف واقعی = حواله‌های قطع‌بوش‌شده (تراکنش ISSUE — شامل حواله تولید و حواله انبار) از دفتر کل تغییرناپذیر.',
   634→    'سرانه استاندارد = تولید واقعی انبار محصول (رسیدهای قطع‌بوش‌شده) × سرانه BOM هر قطعه.',
   635→    'مبنا تولید دوره: ' + (consumedParts.length ? consumedParts.join('، ') : 'تولیدی ثبت نشده است'),
   636→  ];
   637→  if (defaultApplied) notes.push('بازه پیش‌فرض: ۳۰ روز گذشته (بدون فیلتر تاریخ از سوی کاربر).');
   638→  if (missingBomParts.length) {
   639→    notes.push(
   640→      'قطعات تولیدی بدون BOM (سرانه قابل محاسبه نیست): ' +
   641→        missingBomParts.map((pid) => partById.get(pid)?.partNumber ?? pid).join('، '),
   642→    );
   643→  }
   644→
   645→  return table('material-consumption', 'مصرف ماده اولیه در مقابل سرانه استاندارد', columns, rows, meta(f, notes, defaultApplied ? { from: isoDate(period.from!), to: isoDate(period.to!), basisFa: '۳۰ روز گذشته' } : undefined));
   646→}

   648→// ---------------------------------------------------------------------------
   649→// 7) SAPCO material account (kg)
   650→// ---------------------------------------------------------------------------
   651→
   652→export async function sapcoAccount(f: { from?: string; to?: string; materialId?: string }): Promise<ReportTable> {
   653→  const from = f.from ? new Date(`${f.from}T00:00:00.000Z`) : null;
   654→  const to = f.to ? endOfDay(new Date(`${f.to}T00:00:00.000Z`)) : null;
   655→  const periodWhere = {
   656→    ...(f.materialId ? { materialId: f.materialId } : {}),
   657→    ...(from || to ? { occurredAt: { ...(from ? { gte: from } : {}), ...(to ? { lte: to } : {}) } } : {}),
   658→  };
   659→  const beforeWhere = { ...(f.materialId ? { materialId: f.materialId } : {}), ...(from ? { occurredAt: { lt: from } } : {}) };
   660→
   661→  const [periodGroups, openingGroups, materials, balances, contracts] = await Promise.all([
   662→    db.inventoryTransaction.groupBy({
   663→      by: ['materialId', 'type'],
   664→      _sum: { quantity: true, weightKg: true },
   665→      where: { ...periodWhere, materialId: { not: null } },
   666→    }),
   667→    db.inventoryTransaction.groupBy({
   668→      by: ['materialId'],
   669→      _sum: { quantity: true, weightKg: true },
   670→      where: { ...beforeWhere, materialId: { not: null } },
   671→    }),
   672→    db.material.findMany({
   673→      where: f.materialId ? { id: f.materialId } : {},
   674→      select: { id: true, materialCode: true, nameFa: true, baseUnit: { select: { code: true } } },
   675→    }),
   676→    db.stockBalance.groupBy({
   677→      by: ['materialId', 'ownership'],
   678→      _sum: { quantity: true, weightKg: true },
   679→      where: { materialId: { not: null }, ...(f.materialId ? { materialId: f.materialId } : {}) },
   680→    }),
   681→    db.materialContract.findMany({
   682→      where: { status: { in: ['OPEN', 'PARTIALLY_RECEIVED'] }, materialId: f.materialId },
   683→      select: { materialId: true, totalWeightKg: true, transactions: { select: { weightKg: true } } },
   684→    }),
   685→  ]);
   686→
   687→  const openingByMat = new Map<string, D>();
   688→  for (const g of openingGroups) if (g.materialId) openingByMat.set(g.materialId, Q(g._sum.weightKg));
   689→
   690→  const typeByMat = new Map<string, Map<string, D>>();
   691→  for (const g of periodGroups) {
   692→    if (!g.materialId) continue;
   693→    const m = typeByMat.get(g.materialId) ?? new Map<string, D>();
   694→    m.set(g.type, Q(g._sum.weightKg));
   695→    typeByMat.set(g.materialId, m);
   696→  }
   697→
   698→  const balanceOwned = new Map<string, D>();
   699→  const balanceConsignment = new Map<string, D>();
   700→  for (const b of balances) {
   701→    if (!b.materialId) continue;
   702→    if (b.ownership === 'OWNED') balanceOwned.set(b.materialId, Q(b._sum.weightKg));
   703→    else balanceConsignment.set(b.materialId, Q(b._sum.weightKg));
   704→  }
   705→
   706→  const contractRemaining = new Map<string, D>();
   707→  for (const c of contracts) {
   708→    const received = c.transactions.reduce((acc, t) => acc.plus(Q(t.weightKg)), ZERO);
   709→    const total = c.totalWeightKg === null ? ZERO : Q(c.totalWeightKg);
   710→    contractRemaining.set(c.materialId, (contractRemaining.get(c.materialId) ?? ZERO).plus(total.minus(received)));
   711→  }
   712→
   713→  const rows: Record<string, unknown>[] = [];
   714→  for (const m of materials) {
   715→    const types = typeByMat.get(m.id) ?? new Map<string, D>();
   716→    const sumOf = (list: string[]) => list.reduce((acc, t) => acc.plus(types.get(t) ?? ZERO), ZERO);
   717→    const opening = openingByMat.get(m.id) ?? ZERO;
   718→    const receipts = sumOf(['RECEIPT', 'TRANSFER_IN']);
   719→    const consumption = sumOf(CONSUMPTION_TX);
   720→    const adjustments = sumOf(['ADJUSTMENT_IN']).minus(sumOf(['ADJUSTMENT_OUT']));
   721→    const closing = opening.plus(receipts).minus(consumption).plus(adjustments);
   722→
   723→    rows.push({
   724→      materialId: m.id,
   725→      material: `${m.materialCode} — ${m.nameFa}`,
   726→      unit: 'kg',
   727→      openingKg: qty(opening),
   728→      receiptsKg: qty(receipts),
   729→      consumptionKg: qty(consumption),
   730→      adjustmentsKg: qty(adjustments),
   731→      closingKg: qty(closing),
   732→      balanceOwnedKg: qty(balanceOwned.get(m.id) ?? ZERO),
   733→      balanceConsignmentKg: qty(balanceConsignment.get(m.id) ?? ZERO),
   734→      closingMatchesLedger: true,
   735→      openContractRemainingKg: qty(contractRemaining.get(m.id) ?? ZERO),
   736→      hasMovements: types.size > 0 || opening.comparedTo(0) !== 0,
   737→      _drill: drill('sapco_material', m.id, `حساب ${m.materialCode}`),
   738→    });
   739→  }
   740→  rows.sort((a, b) => String(a.material).localeCompare(String(b.material), 'fa'));
   741→
   742→  // reconciliation column: closing must equal signed ledger up to "to"
   743→  const ledgerTo = await ledgerTotalsByDims({ materialId: { not: null }, ...(f.materialId ? { materialId: f.materialId } : {}), ...(to ? { occurredAt: { lte: to } } : {}) });
   744→  const matById = new Map(materials.map((m) => [m.id, m]));
   745→  const ledgerClosing = new Map<string, D>();
   746→  for (const [key, totals] of ledgerTo) {
   747→    const mid = key.split('|')[1];
   748→    if (mid && matById.has(mid)) ledgerClosing.set(mid, (ledgerClosing.get(mid) ?? ZERO).plus(totals.weightKg));
   749→  }
   750→  for (const r of rows) {
   751→    const m = matById.get(String(r.materialId));
   752→    if (!m) continue;
   753→    const ledgerQty = ledgerClosing.get(m.id) ?? ZERO;
   754→    r.closingMatchesLedger = dec(String(r.closingKg) || 0).minus(ledgerQty).abs().comparedTo('0.0001') <= 0;
   755→  }
   756→
   757→  const columns: ReportColumn[] = [
   758→    { key: 'material', labelFa: 'ماده اولیه', kind: 'IDENTIFIER' },
   759→    { key: 'unit', labelFa: 'واحد', kind: 'IDENTIFIER' },
   760→    { key: 'openingKg', labelFa: 'افتتاحیه', kind: 'ACTUAL', unit: 'kg' },
   761→    { key: 'receiptsKg', labelFa: 'دریافتی دوره', kind: 'ACTUAL', unit: 'kg' },
   762→    { key: 'consumptionKg', labelFa: 'مصرف دوره', kind: 'ACTUAL', unit: 'kg' },
   763→    { key: 'adjustmentsKg', labelFa: 'تعدیلات دوره', kind: 'ACTUAL', unit: 'kg' },
   764→    { key: 'closingKg', labelFa: 'پایانی دوره', kind: 'INFO', unit: 'kg' },
   765→    { key: 'balanceOwnedKg', labelFa: 'موجودی فعلی مالی', kind: 'ACTUAL', unit: 'kg' },
   766→    { key: 'balanceConsignmentKg', labelFa: 'موجودی فعلی امانی SAPCO', kind: 'ACTUAL', unit: 'kg' },
   767→    { key: 'closingMatchesLedger', labelFa: 'تطبیق با دفتر کل', kind: 'INFO' },
   768→    { key: 'openContractRemainingKg', labelFa: 'باقی‌مانده قراردادهای باز', kind: 'PLANNED', unit: 'kg' },
   769→    { key: 'hasMovements', labelFa: 'گردش دارد؟', kind: 'INFO' },
   770→    { key: '_drill', labelFa: 'جزئیات منبع', kind: 'INFO' },
   771→  ];
   772→
   773→
   774→  const notes = [
   775→    'همه ارقام کیلوگرمی و از دفتر کل تراکنش‌های قطع‌بوش‌شده است؛ افتتاحیه = جمع دفتر کل قبل از ابتدای دوره.',
   776→    'ستون «پایانی دوره» = افتتاحیه + دریافتی − مصرف + تعدیلات و با جمع دفتر کل تا پایان دوره تطبیق داده می‌شود.',
   777→    'موجودی امانی (SUPPLIER_CONSIGNMENT) جدا از موجودی مالی نشان داده می‌شود (الزام SAPCO).',
   778→  ];
   779→
   780→  return table('sapco-account', 'حساب مواد SAPCO (کیلوگرمی)', columns, rows, meta(f, notes));

   843→export async function scrapReport(f: { from?: string; to?: string; partId?: string; operationId?: string }): Promise<ReportTable> {
   844→  const scraps = await db.productionScrap.findMany({
   845→    where: {
   846→      partId: f.partId,
   847→      operationId: f.operationId,
   848→      scrapDate: { ...(f.from ? { gte: new Date(`${f.from}T00:00:00.000Z`) } : {}), ...(f.to ? { lte: endOfDay(new Date(`${f.to}T00:00:00.000Z`)) } : {}) },
   849→    },
   850→    include: {
   851→      operation: { select: { sequence: true, operation: { select: { code: true, nameFa: true } }, inputQuantity: true, manufacturingOrder: { select: { partId: true, moNumber: true, part: { select: { partNumber: true, nameFa: true } } } } } },
   852→      report: { select: { reportNumber: true } },
   853→    },
   854→    take: 5000,
   855→  });
   856→
   857→  const group = new Map<string, {
   858→    partId: string | null; part: string; operationKey: string; operationLabel: string;
   859→    quantity: D; weight: D; count: number; input: D; reasons: Map<string, D>; docRefs: string[];
   860→  }>();
   861→  for (const s of scraps) {
   862→    const op = s.operation;
   863→    const part = op ? `${op.manufacturingOrder.part.partNumber} — ${op.manufacturingOrder.part.nameFa}` : (s.partId ? 'قطعه (بدون عملیات)' : 'نامشخص');
   864→    const partId = op?.manufacturingOrder.partId ?? s.partId ?? null;
   865→    const operationKey = op ? `${op.operation.code}` : '—';
   866→    const operationLabel = op ? `${op.operation.code} — ${op.operation.nameFa}` : '—';
   867→    const key = `${part}|${operationKey}`;
   868→    const g = group.get(key) ?? {
   869→      partId, part, operationKey, operationLabel,
   870→      quantity: ZERO, weight: ZERO, count: 0, input: ZERO, reasons: new Map<string, D>(), docRefs: [] as string[],
   871→    };
   872→    g.quantity = g.quantity.plus(Q(s.quantity));
   873→    g.weight = g.weight.plus(Q(s.weightKg));
   874→    g.count += 1;
   875→    if (op) g.input = g.input.plus(Q(op.inputQuantity));
   876→    g.reasons.set(s.reason, (g.reasons.get(s.reason) ?? ZERO).plus(Q(s.quantity)));
   877→    if (s.report) g.docRefs.push(s.report.reportNumber);
   878→    group.set(key, g);
   879→  }
   880→
   881→  const rows = [...group.values()]
   882→    .sort((a, b) => a.part.localeCompare(b.part, 'fa') || a.operationKey.localeCompare(b.operationKey))
   883→    .map((g) => ({
   884→      part: g.part,
   885→      operation: g.operationLabel,
   886→      scrapCount: g.count,
   887→      quantity: qty(g.quantity),
   888→      weightKg: g.weight.comparedTo(0) === 0 ? null : qty(g.weight),
   889→      inputQuantity: g.input.comparedTo(0) === 0 ? null : qty(g.input),
   890→      scrapPctOfInput: g.input.comparedTo(0) > 0 ? Math.round(g.quantity.div(g.input).mul(100).toNumber() * 10) / 10 : null,
   891→      reasons: [...g.reasons.entries()].map(([r, q]) => `${r}: ${fmt(q)}`).join('، '),
   892→      sourceReports: [...new Set(g.docRefs)].slice(0, 10).join('، '),
   893→      _drill: drill('scrap_group', `${g.partId ?? ''}|${g.operationKey}|${f.from ?? ''}|${f.to ?? ''}`, `ضایعات ${g.part} / ${g.operationKey}`),
   894→    }));
   895→
   896→  const columns: ReportColumn[] = [
   897→    { key: 'part', labelFa: 'قطعه', kind: 'IDENTIFIER' },
   898→    { key: 'operation', labelFa: 'عملیات', kind: 'IDENTIFIER' },
   899→    { key: 'scrapCount', labelFa: 'تعداد رکورد', kind: 'INFO' },
   900→    { key: 'quantity', labelFa: 'مقدار ضایعات', kind: 'ACTUAL', unit: 'عدد' },
   901→    { key: 'weightKg', labelFa: 'وزن', kind: 'ACTUAL', unit: 'kg' },
   902→    { key: 'inputQuantity', labelFa: 'ورودی عملیات', kind: 'ACTUAL', unit: 'عدد' },
   903→    { key: 'scrapPctOfInput', labelFa: 'سهم از ورودی ٪', kind: 'INFO' },
   904→    { key: 'reasons', labelFa: 'علت‌ها (مقدار)', kind: 'INFO' },
   905→    { key: 'sourceReports', labelFa: 'گزارش‌های منبع', kind: 'IDENTIFIER' },
   906→    { key: '_drill', labelFa: 'جزئیات منبع', kind: 'INFO' },
   907→  ];
   908→
   909→  return table('scrap', 'ضایعات به تفکیک قطعه و عملیات', columns, rows, meta(f, [
   910→    'سهم از ورودی = ضایعات گروه ÷ کل ورودی تغذیه‌شده همان عملیات‌ها (فقط ورودی واقعی ثبت‌شده).',
   911→  ]));
   912→}

   965→export async function productionDelays(f: { partId?: string; customerId?: string; machineId?: string }): Promise<ReportTable> {
   966→  const today = new Date(`${isoDate(new Date())}T00:00:00.000Z`);
   967→  const DAY = 86_400_000;
   968→  const rows: Record<string, unknown>[] = [];
   969→
   970→  // A) customer order lines open past due
   971→  const lateLines = await db.customerOrderLine.findMany({
   972→    where: {
   973→      status: { in: OPEN_LINE_STATUSES },
   974→      partId: f.partId,
   975→      dueDate: { not: null, lt: today },
   976→      order: { status: { in: OPEN_ORDER_STATUSES }, customerId: f.customerId },
   977→    },
   978→    include: {
   979→      order: { select: { orderNumber: true, customer: { select: { nameFa: true } } } },
   980→      part: { select: { partNumber: true, nameFa: true } },
   981→    },
   982→    take: 1000,
   983→  });
   984→  for (const l of lateLines) {
   985→    const open = Q(l.quantity).minus(Q(l.deliveredQuantity));
   986→    if (open.comparedTo(0) <= 0) continue;
   987→    rows.push({
   988→      section: 'ردیف سفارش باز',
   989→      reference: l.order.orderNumber,
   990→      customer: l.order.customer.nameFa,
   991→      part: `${l.part.partNumber} — ${l.part.nameFa}`,
   992→      dueDate: l.dueDate ? isoDate(l.dueDate) : null,
   993→      daysLate: Math.max(1, Math.round((today.getTime() - l.dueDate!.getTime()) / DAY)),
   994→      openQuantity: qty(open),
   995→      hasScheduleDates: null,
   996→      _drill: drill('order_line', l.id, `ردیف سفارش ${l.order.orderNumber}`),
   997→    });
   998→  }
   999→
  1000→  // B) open MOs past due
  1001→  const lateMos = await db.manufacturingOrder.findMany({
  1002→    where: { status: { in: OPEN_MO_STATUSES }, partId: f.partId, dueDate: { not: null, lt: today } },
  1003→    include: { part: { select: { partNumber: true, nameFa: true } } },
  1004→    take: 1000,
  1005→  });
  1006→  for (const mo of lateMos) {
  1007→    rows.push({
  1008→      section: 'دستور کار باز',
  1009→      reference: mo.moNumber,
  1010→      customer: null,
  1011→      part: `${mo.part.partNumber} — ${mo.part.nameFa}`,
  1012→      dueDate: mo.dueDate ? isoDate(mo.dueDate) : null,
  1013→      daysLate: Math.max(1, Math.round((today.getTime() - mo.dueDate!.getTime()) / DAY)),
  1014→      openQuantity: qty(mo.quantity),
  1015→      hasScheduleDates: null,
  1016→      _drill: drill('mo_detail', mo.id, `دستور کار ${mo.moNumber}`),
  1017→    });
  1018→  }
  1019→
  1020→  // C) MOs completed after their due date (real completion timestamp)
  1021→  const completedLate = await db.manufacturingOrder.findMany({
  1022→    where: { status: { in: ['COMPLETED', 'CLOSED'] }, partId: f.partId, dueDate: { not: null }, completedAt: { not: null } },
  1023→    include: { part: { select: { partNumber: true, nameFa: true } } },
  1024→    take: 1000,
  1025→  });
  1026→  for (const mo of completedLate) {
  1027→    const dueEnd = endOfDay(mo.dueDate!);
  1028→    if (mo.completedAt!.getTime() <= dueEnd.getTime()) continue;
  1029→    rows.push({
  1030→      section: 'تکمیل‌شده با تأخیر',
  1031→      reference: mo.moNumber,
  1032→      customer: null,
  1033→      part: `${mo.part.partNumber} — ${mo.part.nameFa}`,
  1034→      dueDate: mo.dueDate ? isoDate(mo.dueDate) : null,
  1035→      daysLate: Math.max(1, Math.ceil((mo.completedAt!.getTime() - dueEnd.getTime()) / DAY)),
  1036→      openQuantity: qty(mo.quantity),
  1037→      hasScheduleDates: null,
  1038→      _drill: drill('mo_detail', mo.id, `دستور کار ${mo.moNumber}`),
  1039→    });
  1040→  }
  1041→
  1042→  // D) operations not completed past their planned end (only where plannedEnd exists)
  1043→  const lateOps = await db.productionOperation.findMany({
  1044→    where: {
  1045→      status: { in: ['PENDING', 'QUEUED', 'IN_PROGRESS', 'PAUSED'] },
  1046→      plannedEnd: { not: null, lt: new Date() },
  1047→      machineId: f.machineId,
  1048→      manufacturingOrder: { partId: f.partId },
  1049→    },
  1050→    include: {
  1051→      operation: { select: { code: true, nameFa: true } },
  1052→      machine: { select: { code: true } },
  1053→      manufacturingOrder: { select: { moNumber: true, part: { select: { partNumber: true, nameFa: true } } } },
  1054→    },
  1055→    take: 1000,
  1056→  });
  1057→  for (const op of lateOps) {
  1058→    rows.push({
  1059→      section: 'عملیات اجرایی عقب‌افتاده',
  1060→      reference: op.manufacturingOrder.moNumber,

  1097→export async function inventoryMovements(f: { from?: string; to?: string; partId?: string; materialId?: string; warehouseId?: string }): Promise<ReportTable> {
  1098→  const { period, defaultApplied } = resolvePeriod(f.from, f.to);
  1099→  const where = {
  1100→    partId: f.partId,
  1101→    materialId: f.materialId,
  1102→    warehouseId: f.warehouseId,
  1103→    ...(period.from || period.to ? { occurredAt: { ...(period.from ? { gte: period.from } : {}), ...(period.to ? { lte: period.to } : {}) } } : {}),
  1104→  };
  1105→  const [groups, parts, materials] = await Promise.all([
  1106→    db.inventoryTransaction.groupBy({
  1107→      by: ['partId', 'materialId', 'type'],
  1108→      _sum: { quantity: true, weightKg: true },
  1109→      _count: { _all: true },
  1110→      where,
  1111→    }),
  1112→    db.part.findMany({ where: f.partId ? { id: f.partId } : {}, select: { id: true, partNumber: true, nameFa: true } }),
  1113→    db.material.findMany({ where: f.materialId ? { id: f.materialId } : {}, select: { id: true, materialCode: true, nameFa: true } }),
  1114→  ]);
  1115→  const partById = new Map(parts.map((p) => [p.id, p]));
  1116→  const matById = new Map(materials.map((m) => [m.id, m]));
  1117→
  1118→  const agg = new Map<string, {
  1119→    item: string; itemKind: 'PART' | 'MATERIAL'; id: string;
  1120→    in: D; out: D; inKg: D; outKg: D; count: number;
  1121→  }>();
  1122→  for (const g of groups) {
  1123→    const isPart = !!g.partId;
  1124→    const id = (g.partId ?? g.materialId)!;
  1125→    const item = isPart
  1126→      ? `${partById.get(id)?.partNumber ?? '?'} — ${partById.get(id)?.nameFa ?? ''}`
  1127→      : `${matById.get(id)?.materialCode ?? '?'} — ${matById.get(id)?.nameFa ?? ''}`;
  1128→    const key = `${isPart ? 'P' : 'M'}|${id}`;
  1129→    const a = agg.get(key) ?? { item, itemKind: (isPart ? 'PART' : 'MATERIAL') as 'PART' | 'MATERIAL', id, in: ZERO, out: ZERO, inKg: ZERO, outKg: ZERO, count: 0 };
  1130→    const sign = TX_SIGN[g.type] ?? 0;
  1131→    const q = Q(g._sum.quantity);
  1132→    const w = Q(g._sum.weightKg);
  1133→    if (sign > 0) { a.in = a.in.plus(q); a.inKg = a.inKg.plus(w); }
  1134→    else if (sign < 0) { a.out = a.out.plus(q.abs()); a.outKg = a.outKg.plus(w.abs()); }
  1135→    a.count += g._count._all;
  1136→    agg.set(key, a);
  1137→  }
  1138→
  1139→  const rows = [...agg.values()]
  1140→    .sort((a, b) => a.item.localeCompare(b.item, 'fa'))
  1141→    .map((a) => ({
  1142→      item: a.item,
  1143→      itemKind: a.itemKind === 'PART' ? 'قطعه' : 'ماده اولیه',
  1144→      transactionCount: a.count,
  1145→      inQuantity: qty(a.in),
  1146→      outQuantity: qty(a.out),
  1147→      netQuantity: qty(a.in.minus(a.out)),
  1148→      inKg: a.inKg.comparedTo(0) === 0 ? null : qty(a.inKg),
  1149→      outKg: a.outKg.comparedTo(0) === 0 ? null : qty(a.outKg),
  1150→      netKg: a.inKg.minus(a.outKg).comparedTo(0) === 0 ? null : qty(a.inKg.minus(a.outKg)),
  1151→      _drill: drill('movement_group', `${a.itemKind}|${a.id}|${period.from ? isoDate(period.from) : ''}|${period.to ? isoDate(period.to) : ''}`, `حرکت ${a.item}`),
  1152→    }));
  1153→
  1154→  const columns: ReportColumn[] = [
  1155→    { key: 'item', labelFa: 'قلم', kind: 'IDENTIFIER' },
  1156→    { key: 'itemKind', labelFa: 'نوع قلم', kind: 'INFO' },
  1157→    { key: 'transactionCount', labelFa: 'تعداد تراکنش', kind: 'ACTUAL' },
  1158→    { key: 'inQuantity', labelFa: 'ورود (مقدار)', kind: 'ACTUAL' },
  1159→    { key: 'outQuantity', labelFa: 'خروج (مقدار)', kind: 'ACTUAL' },
  1160→    { key: 'netQuantity', labelFa: 'خالص (مقدار)', kind: 'INFO' },
  1161→    { key: 'inKg', labelFa: 'ورود (kg)', kind: 'ACTUAL', unit: 'kg' },
  1162→    { key: 'outKg', labelFa: 'خروج (kg)', kind: 'ACTUAL', unit: 'kg' },
  1163→    { key: 'netKg', labelFa: 'خالص (kg)', kind: 'INFO', unit: 'kg' },
  1164→    { key: '_drill', labelFa: 'جزئیات منبع', kind: 'INFO' },
  1165→  ];
  1166→
  1167→  const notes = ['مبنای گزارش دفتر کل تغییرناپذیر تراکنش‌های قطع‌بوش‌شده است (پیش‌نویس‌ها اثر ندارند).'];
  1168→  if (defaultApplied) notes.push('بازه پیش‌فرض: ۳۰ روز گذشته (بدون فیلتر تاریخ از سوی کاربر).');
  1169→
  1170→  return table('inventory-movements', 'حرکت موجودی انبار', columns, rows, meta(f, notes, defaultApplied ? { from: isoDate(period.from!), to: isoDate(period.to!), basisFa: '۳۰ روز گذشته' } : undefined));
  1171→}

  1177→export async function inventoryReconciliation(f: { partId?: string; materialId?: string; warehouseId?: string }): Promise<ReportTable> {
  1178→  const balanceWhere = { partId: f.partId, materialId: f.materialId, warehouseId: f.warehouseId };

  1223→  }

  1225→
  1226→  const columns: ReportColumn[] = [
  1227→    { key: 'item', labelFa: 'قلم', kind: 'IDENTIFIER' },

  1232→    { key: 'ownership', labelFa: 'مالکیت', kind: 'INFO' },
  1233→    { key: 'qualityStatus', labelFa: 'کیفیت', kind: 'INFO' },
  1234→    { key: 'balanceQuantity', labelFa: 'موجودی رسمی', kind: 'ACTUAL' },
